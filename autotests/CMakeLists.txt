# Standalone Google Test suite for deepin-image-viewer (qt-autotest-generator skill).
# Built independently of the application CMake so it does not require the full
# Qt6 Quick/QML toolchain that the application target pulls in. Only the Qt6
# base modules available on the host are used; OCR/ncnn/opencv are resolved via
# pkg-config from the locally extracted development packages.

cmake_minimum_required(VERSION 3.16)
project(deepin_image_viewer_autotests VERSION 6.0.48 LANGUAGES CXX)

set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)

option(ENABLE_COVERAGE "Enable code coverage instrumentation" ON)

# ---- Application sources root ----
set(APP_SRC "${CMAKE_CURRENT_SOURCE_DIR}/../src/src")
set(APP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")

# ---- config.h (normally generated by the application CMake) ----
set(VERSION "6.0.48")
set(PROJECT_VERSION "6.0.48")
configure_file("${APP_ROOT}/config.h.in" "${CMAKE_BINARY_DIR}/config.h" @ONLY)

# ---- Dependencies ----
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Gui Widgets Test Concurrent DBus PrintSupport Svg)
find_package(Dtk6 REQUIRED COMPONENTS Core Widget Gui)
find_package(GTest REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_search_module(OCR_PLUGIN REQUIRED deepin-ocr-plugin-manager)
pkg_check_modules(InferenceEngine REQUIRED ncnn opencv_mobile)

# Allow overriding the OCR include/library dirs when the pkg-config file resolves
# paths incorrectly (e.g. out-of-tree extracted multiarch packages).
if(DEFINED ENV{OCR_INCLUDE_DIRS})
    string(REPLACE ":" ";" _ocr_inc "$ENV{OCR_INCLUDE_DIRS}")
    set(OCR_PLUGIN_INCLUDE_DIRS ${_ocr_inc})
endif()
if(DEFINED ENV{OCR_LIBRARY_DIRS})
    string(REPLACE ":" ";" _ocr_lib "$ENV{OCR_LIBRARY_DIRS}")
    set(OCR_PLUGIN_LIBRARY_DIRS ${_ocr_lib})
endif()

# ---- Include paths (mirror the application) ----
include_directories("${CMAKE_BINARY_DIR}")            # generated config.h
include_directories("${APP_ROOT}/src")
include_directories("${APP_SRC}")
include_directories("${APP_SRC}/unionimage")
include_directories("${APP_SRC}/imagedata")
include_directories("${APP_SRC}/dbus")
include_directories("${APP_SRC}/ocr")
include_directories("${APP_SRC}/printdialog")
include_directories("${APP_SRC}/utils")
include_directories("${APP_SRC}/misc")
include_directories("${APP_SRC}/declarative")
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/3rdparty/stub")
include_directories(${OCR_PLUGIN_INCLUDE_DIRS})
include_directories(${InferenceEngine_INCLUDE_DIRS})
# Some sources include OCR headers as <deepin-ocr-plugin-manager/...>; add the
# parent of the pkg-config include dir so that form resolves too.
foreach(inc ${OCR_PLUGIN_INCLUDE_DIRS})
    get_filename_component(_ocr_parent "${inc}/.." ABSOLUTE)
    include_directories("${_ocr_parent}")
endforeach()

# When Qt6 dev modules (e.g. Svg) are provided out-of-tree, the imported targets may
# point their INTERFACE_INCLUDE_DIRECTORIES at the build-time /usr prefix while the
# headers actually live elsewhere. Allow the caller to inject the real include
# root via the environment.
if(DEFINED ENV{QT_EXTRA_INCLUDE_DIR})
    include_directories($ENV{QT_EXTRA_INCLUDE_DIR})
    # QtSvg headers live in the QtSvg subdirectory; add it so both
    # <QSvgGenerator> and <QtSvg/QSvgGenerator> resolve.
    include_directories($ENV{QT_EXTRA_INCLUDE_DIR}/QtSvg)
endif()

# Allow tests to reach private/protected members.
add_compile_options(-fno-access-control)

# ---- Coverage flags ----
if(ENABLE_COVERAGE)
    add_compile_options(-g -O0 -fno-inline -fprofile-arcs -ftest-coverage)
    add_link_options(--coverage)
endif()

# ---- Application source files (exclude GUI/QML-only translation units) ----
file(GLOB_RECURSE APP_SOURCES CONFIGURE_DEPENDS
    "${APP_SRC}/*.cpp")
# Exclude QML/Quick-backed translation units: they require the Qt6 Quick/Qml
# toolchain and, when linked in, destabilize the offscreen test run (a
# QFileSystemWatcher-based case hangs). Enable them by removing this list and
# linking Qt6::Quick/Qt6::Qml once the QML dev packages are available.
list(REMOVE_ITEM APP_SOURCES
    "${APP_SRC}/declarative/mousetrackitem.cpp"
    "${APP_SRC}/declarative/pathviewrangehandler.cpp"
    "${APP_SRC}/imagedata/imageprovider.cpp"
    "${APP_SRC}/ocr/livetextanalyzer.cpp"
)

# ---- Test sources ----
file(GLOB TEST_SOURCES CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
set(STUB_IMPL "${CMAKE_CURRENT_SOURCE_DIR}/3rdparty/stub/stub-shadow.cpp")

# ---- Test executable ----
add_executable(deepin-image-viewer-autotest
    ${APP_SOURCES}
    ${TEST_SOURCES}
    ${STUB_IMPL}
)
target_link_libraries(deepin-image-viewer-autotest PRIVATE
    GTest::gtest
    Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Concurrent Qt6::DBus Qt6::PrintSupport Qt6::Svg Qt6::Test
    Dtk6::Core Dtk6::Widget Dtk6::Gui
    ${OCR_PLUGIN_LIBRARIES}
    ${InferenceEngine_LIBRARIES}
)
target_link_directories(deepin-image-viewer-autotest PRIVATE
    ${OCR_PLUGIN_LIBRARY_DIRS}
    ${InferenceEngine_LIBRARY_DIRS}
)

# Ensure the runtime can find locally extracted .so files.
set_target_properties(deepin-image-viewer-autotest PROPERTIES
    BUILD_RPATH "${OCR_PLUGIN_LIBRARY_DIRS};${InferenceEngine_LIBRARY_DIRS}")

enable_testing()
add_test(NAME deepin_image_viewer_autotest COMMAND deepin-image-viewer-autotest)

message(STATUS "AUTOTEST: app sources = ${APP_SOURCES}")
message(STATUS "AUTOTEST: test sources = ${TEST_SOURCES}")
